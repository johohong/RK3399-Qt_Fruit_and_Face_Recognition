cmake_minimum_required(VERSION 3.10)
project(FruitDetect2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn objdetect)
find_package(Threads REQUIRED)
if(WIN32)
    set(_mnn_default "${CMAKE_CURRENT_SOURCE_DIR}/third_party/mnn-windows")
else()
    set(_mnn_default /usr/local)
endif()
set(MNN_ROOT "${_mnn_default}" CACHE PATH "Platform-matched MNN installation prefix")
find_path(MNN_INCLUDE_DIR NAMES MNN/Interpreter.hpp PATHS "${MNN_ROOT}/include" NO_DEFAULT_PATH)
find_library(MNN_LIBRARY NAMES MNN PATHS "${MNN_ROOT}/lib" "${MNN_ROOT}/lib64" NO_DEFAULT_PATH)
if(NOT MNN_INCLUDE_DIR OR NOT MNN_LIBRARY)
    message(FATAL_ERROR "MNN missing under MNN_ROOT=${MNN_ROOT}; Linux needs ARM64 libMNN.so, not Windows DLL/lib")
endif()
add_library(FruitMNN UNKNOWN IMPORTED)
set_target_properties(FruitMNN PROPERTIES IMPORTED_LOCATION "${MNN_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MNN_INCLUDE_DIR}")
add_library(FruitCore STATIC src/detector.cpp src/postprocess.cpp src/io.cpp)
target_include_directories(FruitCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(FruitCore PUBLIC FRUIT_WITH_MNN)
target_link_libraries(FruitCore PUBLIC Qt5::Core Qt5::Gui ${OpenCV_LIBS} FruitMNN Threads::Threads)
if(WIN32)
    target_compile_definitions(FruitCore PUBLIC USING_MNN_DLL)
endif()
add_executable(FruitDetect2 src/main.cpp src/mainwindow.cpp src/mainwindow.h src/worker.cpp src/worker.h
    src/photos.cpp src/album.cpp src/album.h src/face_engine.cpp src/face_dialog.cpp src/admin_dialog.h)
target_link_libraries(FruitDetect2 PRIVATE FruitCore Qt5::Widgets Qt5::Concurrent)
add_executable(FruitDetect2Cli src/cli.cpp)
target_link_libraries(FruitDetect2Cli PRIVATE FruitCore)
option(FRUIT_RK3399_TUNING "Enable supplied RK3399 GCC tuning flags for application code only" OFF)
if(FRUIT_RK3399_TUNING)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "FRUIT_RK3399_TUNING requires an AArch64 target")
    endif()
    include(CheckCXXCompilerFlag)
    foreach(flag "-mcpu=cortex-a72.cortex-a53+crypto+crc" "-mtune=cortex-a72.cortex-a53")
        string(MAKE_C_IDENTIFIER "supports_${flag}" flag_test)
        check_cxx_compiler_flag("${flag}" ${flag_test})
        if(NOT ${flag_test})
            message(FATAL_ERROR "Compiler rejects ${flag}; verify board GCC or disable application tuning")
        endif()
        foreach(target FruitCore FruitDetect2 FruitDetect2Cli)
            target_compile_options(${target} PRIVATE "${flag}")
        endforeach()
    endforeach()
endif()
get_filename_component(MNN_LIBRARY_DIR "${MNN_LIBRARY}" DIRECTORY)
foreach(target FruitCore FruitDetect2 FruitDetect2Cli)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
foreach(target FruitDetect2 FruitDetect2Cli)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin" BUILD_RPATH "${MNN_LIBRARY_DIR}" INSTALL_RPATH "${MNN_LIBRARY_DIR};/usr/local/lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
endforeach()
add_custom_target(FruitModels ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin/models"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin/face-models"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/models" "${CMAKE_BINARY_DIR}/bin/models"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/face-models" "${CMAKE_BINARY_DIR}/bin/face-models")
add_dependencies(FruitDetect2 FruitModels)
add_dependencies(FruitDetect2Cli FruitModels)
install(TARGETS FruitDetect2 FruitDetect2Cli RUNTIME DESTINATION bin)
install(DIRECTORY models face-models DESTINATION bin)
message(STATUS "OpenCV ${OpenCV_VERSION}: ${OpenCV_DIR}")
message(STATUS "MNN: ${MNN_LIBRARY}")
message(STATUS "CPU inference; Qt GUI; platform camera backend; no model retraining")

option(FRUIT_BUILD_TESTS "Build offline tests" OFF)
if(FRUIT_BUILD_TESTS)
    enable_testing()
    add_executable(AdminTests tests/admin_test.cpp)
    target_include_directories(AdminTests PRIVATE src)
    target_link_libraries(AdminTests PRIVATE Qt5::Widgets)
    add_executable(PhotoFaceTests tests/photo_face_test.cpp src/face_engine.cpp src/photos.cpp)
    target_link_libraries(PhotoFaceTests PRIVATE FruitCore Qt5::Gui)
    foreach(target AdminTests PhotoFaceTests)
        set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8)
        endif()
    endforeach()
    add_test(NAME admin COMMAND AdminTests)
    add_test(NAME photos_faces COMMAND PhotoFaceTests "${CMAKE_BINARY_DIR}/bin/face-models" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/lena.jpg")
endif()
